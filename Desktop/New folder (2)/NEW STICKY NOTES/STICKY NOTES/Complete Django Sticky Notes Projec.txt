from django.shortcuts import render, get_object_or_404, redirect
from django.contrib import messages
from django.db.models import Q
from django.core.paginator import Paginator
from django.http import HttpResponseRedirect
from django.urls import reverse
from .models import StickyNote
from .forms import StickyNoteForm, SearchForm


def note_list(request):
    """View to display all sticky notes with search functionality."""
    notes = StickyNote.objects.all()
    search_form = SearchForm(request.GET)
    
    # Handle search
    if search_form.is_valid() and search_form.cleaned_data['query']:
        query = search_form.cleaned_data['query']
        notes = notes.filter(
            Q(title__icontains=query) | Q(content__icontains=query)
        )
    
    # Pagination
    paginator = Paginator(notes, 9)  # Show 9 notes per page
    page_number = request.GET.get('page')
    page_obj = paginator.get_page(page_number)
    
    context = {
        'page_obj': page_obj,
        'search_form': search_form,
        'total_notes': notes.count()
    }
    return render(request, 'notes/note_list.html', context)


def note_detail(request, pk):
    """View to display a single sticky note."""
    note = get_object_or_404(StickyNote, pk=pk)
    context = {'note': note}
    return render(request, 'notes/note_detail.html', context)


def note_create(request):
    """View to create a new sticky note."""
    if request.method == 'POST':
        form = StickyNoteForm(request.POST)
        if form.is_valid():
            note = form.save()
            messages.success(request, 'Note created successfully!')
            # Use HttpResponseRedirect to prevent form resubmission on refresh
            return HttpResponseRedirect(reverse('note_detail', args=[note.pk]))
    else:
        form = StickyNoteForm()
    
    context = {'form': form, 'action': 'Create'}
    return render(request, 'notes/note_form.html', context)


def note_edit(request, pk):
    """View to edit an existing sticky note."""
    note = get_object_or_404(StickyNote, pk=pk)
    
    if request.method == 'POST':
        form = StickyNoteForm(request.POST, instance=note)
        if form.is_valid():
            form.save()
            messages.success(request, 'Note updated successfully!')
            # Use HttpResponseRedirect to prevent form resubmission on refresh
            return HttpResponseRedirect(reverse('note_detail', args=[note.pk]))
    else:
        form = StickyNoteForm(instance=note)
    
    context = {'form': form, 'note': note, 'action': 'Edit'}
    return render(request, 'notes/note_form.html', context)


def note_delete(request, pk):
    """View to delete a sticky note."""
    note = get_object_or_404(StickyNote, pk=pk)
    
    if request.method == 'POST':
        note.delete()
        messages.success(request, 'Note deleted successfully!')
        # Use HttpResponseRedirect to prevent form resubmission on refresh
        return HttpResponseRedirect(reverse('note_list'))
    
    context = {'note': note}
    return render(request, 'notes/note_confirm_delete.html', context)
